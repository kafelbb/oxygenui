# Функция автогенерации хедера с байтами ресурсов (Требует CMake 3.30+)
function(embed_resources TARGET_NAME RESOURCE_DIR OUTPUT_HEADER)
    file(GLOB_RECURSE RES_FILES RELATIVE "${RESOURCE_DIR}" "${RESOURCE_DIR}/*")
    
    set(HEADER_CONTENT "// Automatically generated by CMake. Do not modify.\n#pragma once\n#include <cstddef>\n\n")
    
    foreach(RES_FILE ${RES_FILES})
        # Формируем безопасное имя переменной (res/fonts/a.ttf -> res_fonts_a_ttf)
        string(MAKE_C_IDENTIFIER "${RES_FILE}" VAR_NAME)
        
        # Полный путь к исходному файлу
        set(FULL_PATH "${RESOURCE_DIR}/${RES_FILE}")
        
        if(IS_DIRECTORY "${FULL_PATH}")
            continue()
        endif()
        
        # Читаем файл в шестнадцатеричный буфер
        file(READ "${FULL_PATH}" HEX_CONTENT HEX)
        
        # Форматируем шестнадцатеричную строку в массив C-style: 0xaa, 0xbb...
        string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1, " C_ARRAY "${HEX_CONTENT}")
        
        # Считаем размер файла в байтах
        string(LENGTH "${HEX_CONTENT}" HEX_LENGTH)
        math(EXPR BYTE_LENGTH "${HEX_LENGTH} / 2")
        
        # Добавляем данные файла в генерируемый хедер
        string(APPEND HEADER_CONTENT "inline const unsigned char ${VAR_NAME}_dat[] = {\n    ${C_ARRAY}\n};\n")
        string(APPEND HEADER_CONTENT "inline const std::size_t ${VAR_NAME}_len = ${BYTE_LENGTH};\n\n")
    endforeach()
    
    # Записываем сгенерированный текст в файл во время генерации проекта
    file(WRITE "${OUTPUT_HEADER}" "${HEADER_CONTENT}")
endfunction()


# Обновленная функция сборки EXE
function(setup_exe TARGET_NAME SOURCE_FILE)
    set(GENERATED_RES_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/resources.h")
    
    # Генерируем массив байт для ресурсов
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/res")
        embed_resources(${TARGET_NAME} "${CMAKE_CURRENT_SOURCE_DIR}/res" "${GENERATED_RES_HEADER}")
    endif()

    set(EXE_SOURCES ${SOURCE_FILE})

    if(WIN32)
        list(APPEND EXE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/resource.rc")
    endif()

    if(WIN32)
        if(CMAKE_BUILD_TYPE STREQUAL "Release")
            add_executable(${TARGET_NAME} WIN32 ${EXE_SOURCES})
            target_link_options(${TARGET_NAME} PRIVATE "/SUBSYSTEM:WINDOWS" "/ENTRY:mainCRTStartup")
        else()
            add_executable(${TARGET_NAME} ${EXE_SOURCES})
        endif()
    else()
        add_executable(${TARGET_NAME} ${EXE_SOURCES})
    endif()
    
    # Даем таргету доступ к сгенерированному файлу ресурсов
    target_include_directories(${TARGET_NAME} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    
    target_link_libraries(${TARGET_NAME} PRIVATE ass)
    target_link_libraries(${TARGET_NAME} PRIVATE sfml-graphics sfml-window sfml-system)

    if(WIN32)
        target_link_libraries(${TARGET_NAME} PRIVATE opengl32 winmm gdi32 user32 advapi32 shell32)
    endif()

    # ПОСТ-БУЛД КОПИРОВАНИЕ БОЛЬШЕ НЕ НУЖНО — файлы запакованы внутрь exe!
endfunction()

setup_exe(oxyuidemo src/main.cpp)
